extends Node

var events: Array[Dictionary] = []
var m

func _ready() -> void:
	set_meta("status", "starting")
	call_deferred("_run")

func _mark(event: String, detail: Dictionary = {}) -> void:
	events.append({"event":event,"milliseconds":Time.get_ticks_msec(),"detail":detail})
	var file := FileAccess.open("res://DEMO_EVENTS.json", FileAccess.WRITE)
	file.store_string(JSON.stringify(events,"  "))
	set_meta("status",event)

func _frames(count: int) -> void:
	for i in range(count):
		await get_tree().create_timer(1.0/60.0).timeout

func _touch(point: Vector2, pressed: bool) -> void:
	var event := InputEventScreenTouch.new()
	event.index=91
	event.position=point
	event.pressed=pressed
	Input.parse_input_event(event)

func _drag(point: Vector2, previous: Vector2) -> void:
	var event := InputEventScreenDrag.new()
	event.index=91
	event.position=point
	event.relative=point-previous
	Input.parse_input_event(event)

func _tap(point: Vector2) -> void:
	_touch(point,true)
	await _frames(3)
	_touch(point,false)
	await _frames(2)

func _world():
	var found := get_tree().root.find_children("*","OperaCareerWorld2D",true,false)
	return found[0] if not found.is_empty() else null

func _open_task(w) -> bool:
	if not w.task_open:
		var target: Control=w.station_nodes[w.armed_station]
		await _tap(target.get_global_rect().get_center())
	for i in range(900):
		if w.task_open and w.reveal_t <= 0.0:
			await _frames(90)
			return true
		await _frames(1)
	return false

func _capture(label: String) -> void:
	await RenderingServer.frame_post_draw
	get_viewport().get_texture().get_image().save_png("res://captures/"+label+".png")
	_mark("capture",{"label":label})

func _run() -> void:
	m=get_tree().current_scene
	m._apply_quality("speedy")
	get_tree().root.grab_focus()
	m._start_menu_ref()._dismiss_menu()
	m.day_one_active=false
	m._skip_intro()
	m.game="level2"
	m.g["t"]=0.0
	m._enter_castle_interior_now(false)
	await _frames(12)
	m._castle_rooms_ref().show_room("opera_hall",false)
	var routes=m._castle_career_routes_ref()
	if not routes.open_opera_venue():
		_mark("failed_venue")
		return
	await _frames(8)
	await _tap(routes.opera_venue.get_node("OperaLeftElevatorPlaytest").get_global_rect().get_center())
	await _frames(8)
	var menu=routes.opera_venue.job_playtest_menu
	for button: Button in menu.job_buttons:
		if int(button.get_meta("act_index",-1))==12:
			await _tap(button.get_global_rect().get_center())
			break
	await _frames(20)
	var w=_world()
	if w==null:
		_mark("failed_career")
		return
	_mark("real_career_started",{"reward_policy":"existing dev_playtest; owner save isolated"})
	for expected in range(2):
		if not await _open_task(w):
			_mark("failed_task",{"phase":expected})
			return
		var s: Control=w.surface
		if expected==0:
			var center: Vector2=s.get_global_transform_with_canvas()*s._circle_pivot()
			var radius: float=s._circle_min_radius()*1.4
			var last: Vector2=center+Vector2(radius,0)
			_touch(last,true)
			await _frames(2)
			for frame in range(1,241):
				var point: Vector2=center+Vector2.from_angle(float(frame)/240.0*TAU*2.4)*radius
				_drag(point,last)
				last=point
				await _frames(1)
			_touch(last,false)
		else:
			for stroke in range(12):
				if w.phase_advance_pending or w.phase_index!=expected:
					break
				var rect: Rect2=s.get_global_rect()
				var from: Vector2=rect.position+rect.size*Vector2(0.16,0.5)
				var to: Vector2=rect.position+rect.size*Vector2(0.84,0.5)
				var last: Vector2=from
				_touch(last,true)
				await _frames(2)
				for frame in range(1,31):
					var point: Vector2=from.lerp(to,float(frame)/30.0)
					_drag(point,last)
					last=point
					await _frames(1)
				_touch(last,false)
				await _frames(3)
		for wait in range(600):
			if w.phase_index>expected:
				break
			await _frames(1)
		if w.phase_index!=expected+1:
			_mark("failed_prep",{"phase":w.phase_index,"progress":w.phase_progress})
			return
		_mark("prep_completed",{"phase":expected})
	if not await _open_task(w):
		_mark("failed_race_open")
		return
	await _capture("00-race-ready")
	var s: OperaRacerSurface=w.surface
	var anchor:=Vector2(640,550)
	var last: Vector2=anchor
	_touch(anchor,true)
	var seen: Dictionary={}
	for frame in range(4800):
		if s.race_finished:
			break
		var bend: float=s.road_curvature(float(s.kart["s"])+10.0)
		var target: float=-signf(bend)*4.4 if absf(bend)>0.003 else 0.0
		var steer: float=clampf((target-float(s.kart["lat"]))*0.28-float(s.kart.get("latv",0.0))*0.08,-0.55,0.55)
		var point:=anchor+Vector2(steer*OperaRacerSurface.STEERING_RANGE,0)
		_drag(point,last)
		last=point
		await _frames(1)
		var sector: int=s._engine._canvas.current_sector
		if not seen.has(sector) and float(s.kart["s"])>30.0:
			seen[sector]=true
			await _capture("room-%d"%sector)
		if float(s.kart["s"])>=OperaRacerSurface.LAP_DISTANCE*1.15:
			break
	_touch(last,false)
	_mark("interactive_race_ready",{"distance":s.kart["s"],"pearls":s.race_pearls,"visits":seen.keys(),"touch_released":true,"manual_race_state_written":false})
	await _frames(150)
	await _capture("interactive-race-ready")
	_mark("demo_complete_take_over_with_one_finger")
