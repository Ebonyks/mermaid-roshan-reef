const NAVIGATION := preload("res://scripts/stage_navigation_2d.gd")

const ROOT := "res://assets/review/opera_exploration_v1/"

const ROOM_NAMES: Array[String] = ["Arrival lounge", "Music gallery", "Costume gallery", "Rehearsal salon"]

const CRESTS: Array[String] = ["opera_crest_house.png", "opera_crest_singer.png", "opera_crest_ballerina.png", "opera_crest_magician.png"]

func setup(main: ReefMain, original_venue: OperaHouseVenue2D) -> void:
	m = main
	venue = original_venue
	name = "OperaExplorationV1"
	size = VIEW_SIZE
	mouse_filter = Control.MOUSE_FILTER_STOP
	clip_contents = true
	z_index = 100
	venue.accepting_input = false
	completed_mask = m.opera_stars
	navigation = NAVIGATION.new()
	var lanes: Array[PackedVector2Array] = [
		PackedVector2Array([Vector2(215, 560), Vector2(1080, 560)]),
		PackedVector2Array([Vector2(215, 630), Vector2(1080, 630)]),
		PackedVector2Array([Vector2(640, 445), Vector2(640, 630)]),
		PackedVector2Array([Vector2(245, 500), Vector2(245, 630)]),
		PackedVector2Array([Vector2(1035, 500), Vector2(1035, 630)])]
	navigation.configure(lanes)
	world = Node2D.new()
	world.name = "FourScreenWorld"
	add_child(world)
	for room_index: int in range(4):
		_build_room(room_index)
	actor = Sprite2D.new()
	actor.name = "ExploringRoshan"
	actor.z_index = 20
	actor.scale = Vector2.ONE * 0.76
	world.add_child(actor)
	animator = ACTOR_LOOP.new()
	animator.wardrobe_owner = m
	actor.add_child(animator)
	animator.setup_sprite_2d(actor, false, actor)
	marker = Node2D.new()
	marker.name = "TravelDestination"
	marker.z_index = 19
	world.add_child(marker)
	marker.draw.connect(_draw_marker)
	marker.hide()
	_build_hud()
	_show_room(0, Vector2(640, 605))
	m._navigation_push("opera_exploration_v1", self, Callable(self, "_leave_review"))
	m._set_world_controls_enabled(false, "opera_exploration_v1")
	if m.hud_msg != null:
		m.hud_msg.hide()
	m.clear_dialogue()

func _build_room(index: int) -> void:
	var room := Node2D.new()
	room.name = "Screen_" + ROOM_IDS[index]
	room.position = GRID[index] * VIEW_SIZE
	world.add_child(room)
	rooms.append(room)
	var data: Dictionary = JSON.parse_string(FileAccess.get_file_as_string(ROOT + "layout.json")) as Dictionary
	var dimensions: Array = data[ROOM_IDS[index]] as Array
	var art_scale := Vector2(1280.0 / float(dimensions[0]), 720.0 / float(dimensions[1]))
	for column: int in range(2):
		var background := Sprite2D.new()
		background.name = "BackgroundCard_%d" % column
		background.texture = load(ROOT + ROOM_IDS[index] + "_%d.png" % column) as Texture2D
		background.centered = false
		background.position = Vector2(float(column) * 1024.0 * art_scale.x, 0)
		background.scale = art_scale
		background.z_index = -30
		room.add_child(background)
	var ambience: Node2D = AMBIENCE.new()
	ambience.name = "LivingFixtures"
	ambience.room_id = index
	ambience.z_index = 8
	room.add_child(ambience)
	ambiences.append(ambience)
	var chandelier := Sprite2D.new()
	chandelier.name = "AuthoredChandelier"
	chandelier.texture = load("res://assets/flats/castle/interactions_v2/opera_hall_chandelier_sheet.png") as Texture2D
	chandelier.region_enabled = true
	chandelier.region_rect = Rect2(0, 0, 256, 200)
	chandelier.centered = false
	chandelier.position = Vector2(544, 0)
	chandelier.scale = Vector2.ONE * 0.75
	ambience.add_child(chandelier)
	ambience.chandelier = chandelier
	_build_door(room, index, ambience)
	_build_connections(room, index)

func _build_door(room: Node2D, index: int, ambience: Node2D) -> void:
	var door := Sprite2D.new()
	door.name = "PhysicalDoor"
	door.texture = load(ROOT + "doorway.png") as Texture2D
	door.region_enabled = true
	door.region_rect = Rect2(0, 0, 1254, 1254)
	door.position = Vector2(640, 322)
	door.scale = Vector2.ONE * 0.23
	room.add_child(door)
	var shader := Shader.new()
	shader.code = "shader_type canvas_item; uniform float phase = 0.0; void fragment(){ vec2 p=UV; float mask=smoothstep(0.28,0.42,UV.x)*(1.0-smoothstep(0.58,0.72,UV.x))*smoothstep(0.3,0.45,UV.y)*(1.0-smoothstep(0.74,0.85,UV.y)); p.x += sin(UV.y*28.0+phase*0.8)*0.002*mask; COLOR=texture(TEXTURE,p)*COLOR; }"
	var material := ShaderMaterial.new()
	material.shader = shader
	door.material = material
	ambience.curtain = material
	var crest := Sprite2D.new()
	crest.name = "AboveDoorCrest"
	crest.texture = load("res://assets/opera/worlds/ui/crests/" + CRESTS[index]) as Texture2D
	crest.position = Vector2(640, 132)
	crest.scale = Vector2.ONE * 0.35
	crest.z_index = 5
	room.add_child(crest)
	var badge := Sprite2D.new()
	badge.name = "EarnedRoomStar"
	badge.texture = load("res://assets/mg/star.png") as Texture2D
	badge.position = Vector2(688, 143)
	badge.scale = Vector2.ONE * 0.18
	badge.z_index = 6
	room.add_child(badge)
	badges.append(badge)
	objects.append({"object_id": "door_%d" % index, "room": index,
		"rect": Rect2(558, 108, 164, 344), "approach_point": Vector2(640, 445),
		"arrival_radius": ARRIVAL_RADIUS, "action": "career" if index > 0 else "exit",
		"act": ROOM_ACTS[index], "return_target": ROOM_IDS[index]})

func _build_connections(room: Node2D, index: int) -> void:
	var right: bool = index % 2 == 0
	var other: int = index + 1 if right else index - 1
	var x: float = 1035.0 if right else 245.0
	_add_connection(room, index, other, Vector2(x, 560), "›" if right else "‹", "passage")
	var upper: int = index + 2 if index < 2 else index - 2
	x = 245.0 if right else 1035.0
	_add_connection(room, index, upper, Vector2(x, 500), "⌃" if index < 2 else "⌄", "stairs")

func _add_connection(room: Node2D, index: int, destination: int,
		point: Vector2, glyph: String, kind: String) -> void:
	var sign := Label.new()
	sign.name = "PassagePicture_" + kind
	sign.text = glyph
	sign.position = point - Vector2(32, 65)
	sign.size = Vector2(64, 64)
	sign.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	sign.mouse_filter = Control.MOUSE_FILTER_IGNORE
	sign.add_theme_font_size_override("font_size", 60)
	sign.add_theme_color_override("font_color", Color(1.0, 0.91, 0.62))
	sign.add_theme_color_override("font_outline_color", Color(0.35, 0.19, 0.29))
	sign.add_theme_constant_override("outline_size", 5)
	room.add_child(sign)
	objects.append({"object_id": "%s_%d" % [kind, index], "room": index,
		"rect": Rect2(point - Vector2(50, 64), Vector2(100, 104)),
		"approach_point": point, "arrival_radius": ARRIVAL_RADIUS,
		"action": "room", "destination": destination, "return_target": ROOM_IDS[index]})

func _gui_input(event: InputEvent) -> void:
	if busy or paused or job != null:
		return
	if event is InputEventScreenTouch:
		if event.pressed and pointer_owner == -2:
			pointer_owner = event.index
			pointer_position = event.position
		elif not event.pressed and pointer_owner == event.index:
			pointer_owner = -2
			request_destination(event.position)
	elif event is InputEventScreenDrag and pointer_owner == event.index:
		pointer_position = event.position
	elif event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_LEFT:
		if event.pressed and pointer_owner == -2:
			pointer_owner = -1
			pointer_position = event.position
		elif not event.pressed and pointer_owner == -1:
			pointer_owner = -2
			request_destination(event.position)
	elif event is InputEventMouseMotion and pointer_owner == -1:
		pointer_position = event.position
	accept_event()

func request_destination(point: Vector2) -> void:
	if busy or paused or job != null or not is_visible_in_tree():
		return
	cancel_travel()
	var target: Vector2 = navigation.nearest_point(point)
	for object: Dictionary in objects:
		if int(object["room"]) == current_room and (object["rect"] as Rect2).has_point(point):
			pending_object = String(object["object_id"])
			target = object["approach_point"] as Vector2
			break
	route = navigation.route(_local_actor(), target)
	route_cursor = 1 if route.size() > 1 else 0
	marker.position = _room_origin() + target
	marker.show()
	marker.queue_redraw()
	if m != null:
		m._say("talk")

func cancel_travel() -> void:
	route.clear()
	route_cursor = 0
	pending_object = ""
	pointer_owner = -2
	if marker != null:
		marker.hide()

func _process(delta: float) -> void:
	if not is_visible_in_tree() or busy or paused or job != null:
		return
	elapsed += minf(delta, 0.05)
	if marker.visible:
		marker.queue_redraw()
	if route.is_empty():
		return
	var target: Vector2 = _room_origin() + route[route_cursor]
	actor.flip_h = target.x < actor.position.x
	actor.position = actor.position.move_toward(target, SPEED * minf(delta, 0.05))
	if actor.position.distance_to(target) <= 0.5:
		route_cursor += 1
		if route_cursor >= route.size():
			var selected: String = pending_object
			cancel_travel()
			_activate_arrived(selected)

func _activate_arrived(object_id: String) -> void:
	for object: Dictionary in objects:
		if object["object_id"] != object_id or int(object["room"]) != current_room:
			continue
		if _local_actor().distance_to(object["approach_point"] as Vector2) > float(object["arrival_radius"]):
			return
		match String(object["action"]):
			"room":
				_transition_room(int(object["destination"]))
			"career":
				_launch_act(int(object["act"]))
			"exit":
				_leave_review()
		return

func _transition_room(destination: int) -> void:
	busy = true
	travel_count += 1
	transition = create_tween()
	transition.tween_property(fade, "color:a", 1.0, 0.22)
	transition.tween_callback(func() -> void:
		var spawn := Vector2(640, 560)
		if destination / 2 == current_room / 2:
			spawn.x = 300.0 if destination % 2 == 1 else 980.0
		else:
			spawn.x = 245.0 if destination % 2 == 0 else 1035.0
		_show_room(destination, spawn))
	transition.tween_property(fade, "color:a", 0.0, 0.30)
	transition.tween_callback(func() -> void: busy = false)

func _show_room(index: int, spawn: Vector2) -> void:
	current_room = index
	world.position = -_room_origin()
	actor.position = _room_origin() + navigation.nearest_point(spawn)
	for room_index: int in range(4):
		rooms[room_index].visible = room_index == current_room
		ambiences[room_index].enabled = room_index == current_room and not paused
		badges[room_index].visible = ROOM_ACTS[room_index] >= 0 \
			and (completed_mask & (1 << ROOM_ACTS[room_index])) != 0
	title.text = ROOM_NAMES[index]
	locator.queue_redraw()

func _launch_act(act: int) -> void:
	if running_act >= 0 or job != null or act != ROOM_ACTS[current_room]:
		return
	return_position = _local_actor()
	running_act = act
	launches += 1
	hide()
	venue.hide()
	m._castle_rooms_ref().suspend()
	m.clear_dialogue()
	m.game = "opera"
	m.opera_return_room = "opera_hall"
	m.opera_active_act_index = act
	OperaRoutePresentation.suspend(m)
	job = OperaHouse.new()
	m.add_child(job)
	m.opera_game = job
	if not job.start(m, act, Callable(self, "_job_finished"), {}, {"reward_policy": "dev_playtest"}):
		m._navigation_remove("opera_act")
		job.queue_free()
		_job_finished(false)

func _job_finished(completed: bool) -> void:
	if running_act < 0:
		return
	if completed:
		completed_mask |= 1 << running_act
	running_act = -1
	job = null
	m.opera_game = null
	m.opera_active_act_index = -1
	m.opera_return_room = ""
	m.opera_plot_context = ""
	m._restore_opera_route_state("opera_hall")
	venue.open(m.opera_stars)
	venue.accepting_input = false
	m.clear_dialogue()
	if m.hud_msg != null:
		m.hud_msg.hide()
	_show_room(current_room, return_position)
	show()
	m._navigation_push("opera_exploration_v1", self, Callable(self, "_leave_review"))

func _toggle_pause() -> void:
	paused = not paused
	cancel_travel()
	animator.set_paused(paused)
	for ambience: Node2D in ambiences:
		ambience.enabled = not paused
	pause_button.text = "▶" if paused else "Ⅱ"

func _notification(what: int) -> void:
	if what == NOTIFICATION_APPLICATION_FOCUS_OUT and is_node_ready():
		cancel_travel()
		if not paused and job == null:
			_toggle_pause()

func _leave_review() -> void:
	if job != null or busy:
		return
	cancel_travel()
	m._navigation_remove("opera_exploration_v1")
	m._set_world_controls_enabled(true, "opera_exploration_v1")
	venue.accepting_input = true
	venue._go_back()
	queue_free()

func _local_actor() -> Vector2:
	return actor.position - _room_origin()

func _room_origin() -> Vector2:
	return GRID[current_room] * VIEW_SIZE

func _draw_marker() -> void:
	var radius: float = 13.0 + sin(elapsed * 4.0) * 2.0
	marker.draw_arc(Vector2.ZERO, radius, 0, TAU, 32, Color(1, 0.89, 0.47, 0.90), 2.5, true)
	marker.draw_circle(Vector2.ZERO, 4, Color(1, 0.95, 0.77, 0.9))

func _draw_locator() -> void:
	for index: int in range(4):
		var rect := Rect2(GRID[index] * Vector2(47, 29), Vector2(42, 25))
		locator.draw_style_box(_map_style(index == current_room), rect)

func _map_style(selected: bool) -> StyleBoxFlat:
	var style := StyleBoxFlat.new()
	style.bg_color = Color(1, 0.86, 0.48, 0.94) if selected else Color(0.25, 0.17, 0.28, 0.65)
	style.corner_radius_top_left = 6
	style.corner_radius_top_right = 6
	style.corner_radius_bottom_left = 6
	style.corner_radius_bottom_right = 6
	return style

func review_snapshot() -> Dictionary:
	var ambience: Node2D = ambiences[current_room]
	return {"room": ROOM_IDS[current_room], "grid": GRID[current_room],
		"position": _local_actor(), "travelling": not route.is_empty(),
		"pending": pending_object, "completed_mask": completed_mask,
		"running_act": running_act, "launches": launches, "transitions": travel_count,
		"paused": paused, "ambient_phase": ambience.phase,
		"chandelier_frame": int(ambience.phase * 2.0) % 8,
		"actor": animator.animation_state(), "draft_native_resolution": true}
