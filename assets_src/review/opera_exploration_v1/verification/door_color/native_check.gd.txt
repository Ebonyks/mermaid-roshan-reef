extends Node
const OUT := "C:/Users/Peter/Documents/mermaid-roshan-reef/tmp/opera-four-screen-v1-20261031/assets_src/review/opera_exploration_v1/verification/door_color/"
var done := false
func _ready() -> void:
	call_deferred("_run")
func _run() -> void:
	var room: Node = get_tree().current_scene.find_child("OperaExplorationV1",true,false)
	room._set_paused(true)
	var signs: Array[Dictionary] = []
	for child: Node in room.world.get_children():
		if child is Sprite2D and child.has_meta("complete_painted_crest"):
			var sprite: Sprite2D = child as Sprite2D
			var image: Image = sprite.texture.get_image()
			signs.append({"name":str(sprite.name),"texture":sprite.texture.resource_path,"size":str(sprite.texture.get_size()),"scale":str(sprite.scale),"visible":sprite.visible,"corner_alpha":image.get_pixel(0,0).a,"position":str(sprite.position)})
	var layer := CanvasLayer.new()
	layer.layer = 120
	add_child(layer)
	var base: Sprite2D = room.world.get_node("RestoredOriginalScreen_1_0") as Sprite2D
	for index: int in range(3):
		var sprite := Sprite2D.new()
		sprite.texture = base.texture
		sprite.centered = false
		sprite.position = Vector2(20+424*index,220)
		sprite.scale = Vector2.ONE*0.35
		if index==0:
			var plain := CanvasItemMaterial.new()
			plain.light_mode = CanvasItemMaterial.LIGHT_MODE_UNSHADED
			sprite.material = plain
		else:
			var material: ShaderMaterial = base.material.duplicate(true) as ShaderMaterial
			material.set_shader_parameter("motion_enabled",false)
			material.set_shader_parameter("apply_palette",false)
			if index==2:
				var negative := Shader.new()
				negative.code = material.shader.code.replace("pixel * vertex_tint","pixel * COLOR")
				material.shader = negative
			sprite.material = material
		layer.add_child(sprite)
	await RenderingServer.frame_post_draw
	await get_tree().process_frame
	await RenderingServer.frame_post_draw
	get_viewport().get_texture().get_image().save_png(OUT+"texture_once_control.png")
	layer.queue_free()
	await get_tree().process_frame
	await RenderingServer.frame_post_draw
	get_viewport().get_texture().get_image().save_png(OUT+"after_foyer.png")
	room.set_review_overview(true)
	await get_tree().process_frame
	await RenderingServer.frame_post_draw
	get_viewport().get_texture().get_image().save_png(OUT+"after_overview.png")
	room.set_review_overview(false)
	room._set_paused(false)
	var result: Dictionary = {"signs":signs,"sign_count":signs.size(),"new_room_type":room.get_class(),"save_hash":room.m.save_data.hash(),"stars":room.m.opera_stars,"pearls":room.m.pearl_count,"snapshot":room.review_snapshot(),"texture_control":{"panels":"left unshaded source; middle fixed shader palette/motion off; right deliberately restored texture-squared negative control","native_scale":float(get_viewport().get_texture().get_width())/1280.0,"logical_x":[20,444,868],"logical_y":220,"sprite_scale":0.35,"texture_patch":[310,102,473,151]},"engine":Engine.get_version_info(),"renderer":RenderingServer.get_current_rendering_method()}
	var file: FileAccess = FileAccess.open(OUT+"native_checks.json",FileAccess.WRITE)
	file.store_string(JSON.stringify(result,"\t")+"\n")
	done = true
	print("OPERA_DOOR_COLOR|NATIVE_CAPTURE_DONE")
