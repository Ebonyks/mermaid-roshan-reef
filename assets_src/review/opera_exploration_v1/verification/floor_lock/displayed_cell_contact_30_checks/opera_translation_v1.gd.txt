func _build_original_house() -> void:
	var art_root: String = TILED_DETAIL if FileAccess.file_exists(TILED_DETAIL+"layout.json") else RESTORED

	tiled_background = bool(layout.get("native_2k_per_screen",false))
	var pieces: Array = (layout["cards"] as Array).duplicate()
	pieces.append_array(layout.get("objects",[]) as Array)
	for value: Variant in pieces:
		var info: Dictionary = value as Dictionary
		var rect: Array = info["source_rect"] as Array
		var reference: Array = info.get("reference_size",native_size) as Array
		var reference_size := Vector2(float(reference[0]),float(reference[1]))
		var card := Sprite2D.new()
		card.name = String(info.get("id","RestoredOriginalScreen_%d_%d" % [int(int(rect[1])>0),int(int(rect[0])>0)]))
		card.texture = load(art_root+String(info["path"])) as Texture2D
		card.centered = false
		card.region_enabled = true
		card.region_filter_clip_enabled = true
		card.region_rect = Rect2(0,0,float(rect[2]),float(rect[3]))
		card.position = Vector2(float(rect[0]),float(rect[1]))/reference_size*WORLD_SIZE
		card.scale = WORLD_SIZE/reference_size
		card.z_index = 8 if String(info.get("owner","")) in ["upper_balcony","middle_balcony"] else int(info.get("z_index",-30))

func _gui_input(event: InputEvent) -> void:
	if m==null or player_root==null:
		return
	if paused or job != null or overview or busy:
		return
	if event is InputEventScreenTouch:
		if event.pressed and pointer_owner == -2:
			pointer_owner = event.index
			pointer_position = event.position
			_preview_pointer(event.position)
		elif not event.pressed and pointer_owner == event.index:
			pointer_owner = -2
			request_destination(event.position+camera_position)
	elif event is InputEventScreenDrag and pointer_owner == event.index:
		pointer_position = event.position
		_preview_pointer(event.position)
	elif event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_LEFT:
		if event.pressed and pointer_owner == -2:
			pointer_owner = -1
			pointer_position = event.position
			_preview_pointer(event.position)
		elif not event.pressed and pointer_owner == -1:
			pointer_owner = -2
			request_destination(event.position+camera_position)
	elif event is InputEventMouseMotion and pointer_owner == -1:
		pointer_position = event.position
		_preview_pointer(event.position)
	accept_event()

func request_destination(point: Vector2) -> void:
	if paused or job != null or overview or busy or not is_visible_in_tree():
		return
	cancel_travel()
	if not point.is_finite():
		return
	camera_hold = false
	var target: Vector2 = floor_navigation.nearest_point(point)
	for object: Dictionary in objects:
		if (object["rect"] as Rect2).has_point(point):
			pending_object = String(object["object_id"])
			target = object["approach_point"] as Vector2
			if object["action"]=="lift":
				var nearest: int = _nearest_floor(player_root.position.y)
				target.y = FLOOR_Y[nearest]
				object["approach_point"] = target
				var destination: int = _nearest_floor(point.y)
				object["destination_floor"] = destination if destination!=nearest else (nearest+2)%3
			break
	route = _plan_floor_route(player_root.position,target)
	route_cursor = 1 if route.size()>1 else 0
	marker.position = target
	marker.show()
	marker.queue_redraw()
	m._say("roshan","talk")

func cancel_travel() -> void:
	if riding_lift and player_root != null:
		# Interrupt only at a real landing, never leave Roshan suspended in a shaft.
		player_root.position.y = FLOOR_Y[_nearest_floor(player_root.position.y)]
		riding_lift = false
	_recover_floor_position()
	route.clear()
	route_cursor = 0
	pending_object = ""
	pointer_owner = -2
	if marker != null:
		marker.hide()

func _process(delta: float) -> void:
	if m==null or player_root==null:
		return
	if not is_visible_in_tree() or paused or job != null or overview:
		return
	if not player_root.position.is_finite():
		riding_lift = false
	elapsed += minf(delta,0.05)
	if not riding_lift and _recover_floor_position():
		cancel_travel()
	if not route.is_empty() and (route_cursor < 0 or route_cursor >= route.size()):
		cancel_travel()
	if not route.is_empty():
		var target: Vector2 = route[route_cursor]
		var source: Vector2 = route[maxi(0,route_cursor-1)]
		var lift_leg: bool = _is_lift_leg(source,target)
		if lift_leg and not riding_lift:
			riding_lift = true
			lift_rides += 1
			ambience.lift_burst = 2.8
			_play_chime()
		actor.flip_h = target.x < player_root.position.x
		player_root.position = player_root.position.move_toward(target,SPEED*minf(delta,0.05))
		if lift_leg:
			player_root.position.x = source.x
		elif _recover_floor_position():
			cancel_travel()
			_update_camera(delta)
			motion_source.position = player_root.position
			return
		if player_root.position.distance_to(target)<=0.5:
			player_root.position = target
			riding_lift = false
			last_floor_position = target
			route_cursor += 1
			if route_cursor>=route.size():
				var selected: String = pending_object
				cancel_travel()
				_activate_arrived(selected)
	if not riding_lift:
		last_floor_position = player_root.position
	_ground_actor_to_floor()
	_update_camera(delta)
	motion_source.position = player_root.position
	if marker.visible:
		marker.queue_redraw()

func _sprite_foot_bottom(sprite: Sprite2D) -> float:
	var sample: Rect2 = RoshanSpriteFrames.sampled_rect_2d(sprite)
	var texture_id: int = sprite.texture.get_instance_id()
	var key: String = "%d:%s" % [texture_id,str(sample)]
	if not floor_alpha_cache.has(key):
		if not floor_alpha_images.has(texture_id):
			var image: Image = sprite.texture.get_image()
			if image.is_compressed():
				image.decompress()
			floor_alpha_images[texture_id] = image
		var image: Image = floor_alpha_images[texture_id] as Image
		floor_alpha_cache[key] = float(image.get_region(Rect2i(sample)).get_used_rect().end.y)
	return sprite.get_rect().position.y+float(floor_alpha_cache[key])

func _ground_actor_to_floor() -> void:
	# Read-only alpha measurements retain every approved pose and pixel. The
	# gameplay root owns the foot; torso-anchor and idle-breath offsets cannot
	# detach the visible silhouette from its painted floor contact.
	actor.position.y = -_sprite_foot_bottom(actor)*actor.scale.y
	var transition: Node = actor.get_node_or_null("TemporalSpriteTransition")
	if transition != null:
		var ghost: Sprite2D = transition.transition_ghost()
		if ghost != null and ghost.texture != null:
			ghost.position.y = -actor.position.y/actor.scale.y-_sprite_foot_bottom(ghost)*ghost.scale.y

func _walking_route(source: Vector2, target: Vector2) -> PackedVector2Array:
	if source.distance_to(target)<=0.05:
		return PackedVector2Array([source])
	return floor_navigation.route(source,target)

func _plan_floor_route(source: Vector2, target: Vector2) -> PackedVector2Array:
	var direct: PackedVector2Array = _walking_route(source,target)
	if not direct.is_empty():
		return direct
	var best := PackedVector2Array()
	var best_distance := INF
	# Only landings join the walking network to a lift. A shaft crossing a
	# staircase in the painting must never become an accidental mid-air exit.
	for shaft_x: float in [438.0,2120.0]:
		for entry_y: float in FLOOR_Y:
			var entry := Vector2(shaft_x,entry_y)
			var approach: PackedVector2Array = _walking_route(source,entry)
			if approach.is_empty():
				continue
			for exit_y: float in FLOOR_Y:
				if entry_y==exit_y:
					continue
				var landing := Vector2(shaft_x,exit_y)
				var departure: PackedVector2Array = _walking_route(landing,target)
				if departure.is_empty():
					continue
				var candidate: PackedVector2Array = approach.duplicate()
				candidate.append(landing)
				candidate.append_array(departure.slice(1))
				var distance := 0.0
				for index: int in range(1,candidate.size()):
					distance += candidate[index-1].distance_to(candidate[index])
				if distance<best_distance:
					best = candidate
					best_distance = distance
	return best

func _is_lift_leg(source: Vector2, target: Vector2) -> bool:
	if absf(source.x-target.x)>0.05 or absf(source.y-target.y)<0.05:
		return false
	if absf(source.x-438.0)>0.05 and absf(source.x-2120.0)>0.05:
		return false
	return absf(source.y-FLOOR_Y[_nearest_floor(source.y)])<0.05 \
		and absf(target.y-FLOOR_Y[_nearest_floor(target.y)])<0.05

func _recover_floor_position() -> bool:
	if floor_navigation == null or player_root == null:
		return false
	if player_root.position.is_finite() and floor_navigation.contains_point(player_root.position):
		last_floor_position = player_root.position
		return false
	# Match Castle's explicit floor-contact contract. Never award or activate
	# a pending object while recovering from an invalid position.
	player_root.position = last_floor_position
	if not player_root.position.is_finite() or not floor_navigation.contains_point(player_root.position):
		player_root.position = Vector2(1280,1380)
	floor_recoveries += 1
	camera_hold = false
	camera_initialized = false
	return true

func _update_camera(delta: float) -> void:
	if camera_hold:
		world.position = -camera_position
		map_control.queue_redraw()
		return
	var desired := Vector2(clampf(player_root.position.x-640.0,0,1280),
		clampf(player_root.position.y-430.0,0,694))
	if not camera_initialized:
		camera_position = desired
		camera_initialized = true
	else:
		camera_position = camera_position.lerp(desired,1.0-exp(-7.0*delta))
	world.position = -camera_position
	if map_control != null:
		map_control.queue_redraw()

func _activate_arrived(object_id: String) -> void:
	for object: Dictionary in objects:
		if object["object_id"] == object_id:
			if player_root.position.distance_to(object["approach_point"] as Vector2)<=float(object["arrival_radius"]):
				match String(object["action"]):
					"career":
						_open_curtain(int(object["act"]))
					"lift":
						_ride_lift(object)
					"bell", "tea":
						_respond_to_prop(String(object["action"]))
			return

func _nearest_floor(y: float) -> int:
	var nearest := 0
	for index: int in range(1,3):
		if absf(FLOOR_Y[index]-y)<absf(FLOOR_Y[nearest]-y):
			nearest = index
	return nearest

func _preview_pointer(point: Vector2) -> void:
	if not point.is_finite():
		return
	marker.position = floor_navigation.nearest_point(point+camera_position)
	marker.show()
	marker.queue_redraw()

func _ride_lift(object: Dictionary) -> void:
	var target: Vector2 = object["approach_point"] as Vector2
	target.y = FLOOR_Y[int(object["destination_floor"])]
	route = PackedVector2Array([player_root.position,target])
	route_cursor = 1
	pending_object = ""
	marker.position = target
	marker.show()

func _respond_to_prop(action: String) -> void:
	prop_responses += 1
	if action=="tea":
		ambience.tea_burst = 3.0
	else:
		ambience.bell_burst = 2.0
	_play_chime()

func _play_chime() -> void:
	sound.stream = load("res://assets/audio/chime.ogg") as AudioStream
	sound.play()

func _open_curtain(act: int) -> void:
	busy = true
	var row: int = ACTS.find(act)
	opened_curtain_index = row*4+ACT_COLUMNS[row]
	for material: ShaderMaterial in ambience.materials:
		material.set_shader_parameter("selected_door",Vector2(DOOR_X[ACT_COLUMNS[row]],float(row)))
	_play_chime()
	door_tween = create_tween()
	door_tween.tween_method(_set_door_open,0.0,1.0,0.65)
	door_tween.tween_callback(func() -> void:
		busy = false
		_launch_act(act))

func _set_door_open(value: float) -> void:
	for material: ShaderMaterial in ambience.materials:
		material.set_shader_parameter("door_open",value)
	for index: int in range(ambience.curtains.size()):
		ambience.curtains[index].opening = value if index==opened_curtain_index else 0.0

func _launch_act(act: int) -> void:
	if running_act>=0 or job!=null or not ACTS.has(act):
		return
	return_position = player_root.position
	return_camera = camera_position
	running_act = act
	launches += 1
	hide()
	venue.hide()
	m._castle_rooms_ref().suspend()
	m.clear_dialogue()
	m.game = "opera"
	m.opera_return_room = "opera_hall"
	m.opera_active_act_index = act
	OperaRoutePresentation.suspend(m)
	job = OperaHouse.new()
	m.add_child(job)
	m.opera_game = job
	if not job.start(m,act,Callable(self,"_job_finished"),{},{"reward_policy":"dev_playtest"}):
		m._navigation_remove("opera_act")
		job.queue_free()
		_job_finished(false)

func _job_finished(completed: bool) -> void:
	if running_act<0:
		return
	if completed:
		completed_mask |= 1<<running_act
		ambience.celebration_position = badges[ACTS.find(running_act)].position
		ambience.celebration = 2.5
	running_act = -1
	job = null
	m.opera_game = null
	m.opera_active_act_index = -1
	m.opera_return_room = ""
	m.opera_plot_context = ""
	m._restore_opera_route_state("opera_hall")
	venue.open(m.opera_stars)
	venue.accepting_input = false
	m.clear_dialogue()
	player_root.position = return_position
	_recover_floor_position()
	camera_position = return_camera
	camera_hold = true
	world.position = -camera_position
	motion_source.position = player_root.position
	_refresh_badges()
	_set_door_open(0.0)
	show()
	m._navigation_push("opera_exploration_v1",self,Callable(self,"_leave_review"))

func _refresh_badges() -> void:
	for row: int in range(3):
		badges[row].visible = (completed_mask & (1<<ACTS[row]))!=0

func _toggle_pause() -> void:
	focus_paused = false
	_set_paused(not paused)

func _set_paused(value: bool) -> void:
	paused = value
	if paused and door_tween != null and door_tween.is_valid():
		door_tween.kill()
		busy = false
		_set_door_open(0.0)
	cancel_travel()
	animator.set_paused(paused)
	ambience.enabled = not paused
	pause_button.text = "▶" if paused else "||"

func _notification(what: int) -> void:
	if what == NOTIFICATION_APPLICATION_FOCUS_OUT and is_node_ready():
		cancel_travel()
		if not paused and job == null:
			_set_paused(true)
			focus_paused = true
	elif what == NOTIFICATION_APPLICATION_FOCUS_IN and is_node_ready() and focus_paused:
		focus_paused = false
		_set_paused(false)

func _leave_review() -> void:
	if job!=null:
		return
	cancel_travel()
	m._navigation_remove("opera_exploration_v1")
	m._set_world_controls_enabled(true,"opera_exploration_v1")
	venue.accepting_input = true
	venue._go_back()
	queue_free()

func _draw_marker() -> void:
	marker.draw_arc(Vector2.ZERO,14.0+sin(elapsed*4.0)*2.0,0,TAU,32,Color(1,0.9,0.5,0.9),2.5,true)
	marker.draw_circle(Vector2.ZERO,4,Color(1,0.96,0.76,0.9))

	map_control.draw_circle(player_root.position/WORLD_SIZE*Vector2(144,81),3,Color(0.5,1,1))
