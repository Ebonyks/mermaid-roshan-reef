def source(path):
    p = ROOT / path
    if path not in sources:
        b = p.read_bytes()
        try:
            baseline = subprocess.check_output(['git','show',f'{BASE}:{path}'], cwd=ROOT)
            same = sha(b) == sha(baseline)
        except subprocess.CalledProcessError:
            same = False
        sources[path] = dict(path=path, sha256=sha(b), bytes=len(b), equals_baseline=same,
                             immutable_baseline_url=f'{REPO}/blob/{BASE}/{path}')
        if p.suffix.lower() == '.png':
            im = Image.open(p)
            sources[path].update(dimensions=list(im.size), mode=im.mode,
                license_provenance='Existing project runtime artwork; original ASSET_LICENSES.md remains authoritative. Review reuse grants no new acceptance.')
    used.add(path)
    return p

def paste(im, art, box):
    x,y,w,h = box
    a = art.copy()
    a.thumbnail((int(w), int(h)), Image.Resampling.LANCZOS)
    im.alpha_composite(a, (int(x + (w-a.width)/2), int(y + (h-a.height)/2)))

def tiled(template, cols, rows, w=1280, h=720):
    im = Image.new('RGBA',(w,h))
    for y in range(rows):
        for x in range(cols):
            p = source(template.format(x=x,y=y))
            a = Image.open(p).convert('RGBA').resize((w//cols,h//rows),Image.Resampling.LANCZOS)
            im.alpha_composite(a,(x*w//cols,y*h//rows))
    return im

def line(d,points,color='#12b6a1',width=9):
    d.line(points,fill=color,width=width,joint='curve')
    if len(points)>1:
        x,y=points[-1];a=math.atan2(y-points[-2][1],x-points[-2][0])
        d.polygon([(x,y),(x-25*math.cos(a-.55),y-25*math.sin(a-.55)),(x-25*math.cos(a+.55),y-25*math.sin(a+.55))],fill=color)

def ring(d,x,y,r=38):
    d.ellipse((x-r,y-r,x+r,y+r),outline='#12b6a1',width=8)
    d.ellipse((x-7,y-7,x+7,y+7),fill='#12b6a1')

def wrap(d,text,xy,font,width,fill,spacing=5):
    x,y=xy; words=text.split(); rows=[]; row=''
    for word in words:
        s=(row+' '+word).strip()
        if d.textlength(s,font=font)>width and row: rows.append(row);row=word
        else:row=s
    if row:rows.append(row)
    for row in rows:
        d.text((x,y),row,font=font,fill=fill);y+=font.size+spacing
    return y
