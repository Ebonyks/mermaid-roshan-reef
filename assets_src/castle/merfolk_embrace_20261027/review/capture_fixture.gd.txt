extends SceneTree

var output: String = OS.get_environment("PAINTING_CAPTURE_OUT")

func frames(count: int) -> void:
	for index: int in range(count):
		await process_frame

func capture(label: String) -> void:
	await frames(8)
	await RenderingServer.frame_post_draw
	var image: Image = root.get_viewport().get_texture().get_image()
	var result: Error = image.save_png(output.path_join(label + ".png"))
	assert(result == OK)

func run() -> void:
	assert(DisplayServer.get_name() != "headless")
	root.mode = Window.MODE_WINDOWED
	root.size = Vector2i(int(OS.get_environment("PAINTING_CAPTURE_WIDTH")), 720)
	assert(RenderingServer.get_current_rendering_method() == "mobile")
	DirAccess.make_dir_recursive_absolute(output)
	var scene: PackedScene = load("res://scenes/main.tscn") as PackedScene
	var main: Node = scene.instantiate()
	root.add_child(main)
	await frames(3)
	main.call("_start_menu_ref").call("_dismiss_menu")
	main.call("_launch_from_start_menu", false)
	main.call("_apply_quality", "speedy")
	main.call("_skip_intro")
	await frames(2)
	main.set("pearl_count", 10)
	for friend: Dictionary in main.get("friends") as Array:
		friend["found"] = true
		friend["won"] = true
	main.set("trophies", 5)
	main.set("level2_done_once", true)
	main.call("_enter_level2_now", true, false, false)
	await frames(12)
	main.call("_enter_castle_interior_now", false)
	await frames(18)
	assert(not bool(main.get("start_menu_active")))
	assert(root.get_viewport().get_visible_rect().size == Vector2(int(OS.get_environment("PAINTING_CAPTURE_WIDTH")), 720))
	var rooms: RefCounted = main.call("_castle_rooms_ref") as RefCounted
	main.set("day_one_active", false)
	rooms.call("show_room", "royal_bedroom", false)
	await frames(8)
	var layer: Node2D = main.get("castle_room_mid_layer") as Node2D
	var painting: Sprite2D = layer.get_node("MerfolkEmbracePainting") as Sprite2D
	assert(painting.texture.get_size() == Vector2(448, 640))
	assert(painting.material == null)
	assert(painting.modulate == Color.WHITE)
	await capture("royal_bedroom")
	rooms.call("show_room", "family_gallery", false)
	await frames(8)
	assert(layer.find_child("MerfolkEmbracePainting", false, false) == null)
	rooms.call("show_room", "royal_bedroom", false)
	await frames(8)
	assert(layer.get_child_count() == 1)
	await capture("royal_bedroom_reentry")
	print("PAINTING_CAPTURE|PASS|Mobile|", root.get_viewport().get_visible_rect().size, "|room leave/reentry one painting")
	main.queue_free()
	await frames(4)
	quit()

func _init() -> void:
	call_deferred("run")
