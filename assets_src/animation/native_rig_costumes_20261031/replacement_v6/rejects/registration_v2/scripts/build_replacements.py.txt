No new painting: declared piecewise affine registration, source extraction and masks.
The original clothing is never composited below selected clothing.
"""

def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def poly(points):
 im=Image.new('L',(1024,1024));ImageDraw.Draw(im).polygon(points,fill=255);return np.array(im)>0
def warp(image,target,source):
 from scipy.ndimage import map_coordinates
 points=np.c_[x.ravel(),y.ravel()];q=LinearNDInterpolator(np.array(target,float),np.array(source,float),fill_value=-100)(points).reshape(1024,1024,2)
 bad=q[:,:,0]<-50
 if bad.any():q[bad]=NearestNDInterpolator(np.array(target,float),np.array(source,float))(points.reshape(1024,1024,2)[bad])
 a=np.array(image);out=np.stack([map_coordinates(a[:,:,i].astype(float),[q[:,:,1],q[:,:,0]],order=1,mode='constant',cval=0) for i in range(4)],axis=-1)
 return np.uint8(np.clip(out,0,255))
