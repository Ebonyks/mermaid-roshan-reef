def warp(image,target,source,fabric_polygon):
 from scipy.ndimage import map_coordinates
 points=np.c_[x.ravel(),y.ravel()];q=LinearNDInterpolator(np.array(target,float),np.array(source,float),fill_value=-100)(points).reshape(1024,1024,2)
 bad=q[:,:,0]<-50
 if bad.any():q[bad]=NearestNDInterpolator(np.array(target,float),np.array(source,float))(points.reshape(1024,1024,2)[bad])
 roi=(x>390)&(x<675)&(y>320)&(y<850)
 q[roi]=RBFInterpolator(np.array(target,float),np.array(source,float),kernel='thin_plate_spline',smoothing=3)(points.reshape(1024,1024,2)[roi])
 a=np.array(image)
 # Transparent/partial-alpha generation border RGB must never become opaque
 # garment paint when the registered garment adopts the target contour.
 source_mask=Image.new("L",image.size);ImageDraw.Draw(source_mask).polygon(fabric_polygon,fill=255)
 valid=(a[:,:,3]>240)&(np.array(source_mask)>0)
 nearest=distance_transform_edt(~valid,return_distances=False,return_indices=True)
 rgb=a[nearest[0],nearest[1],:3];a[:,:,:3]=rgb
 out=np.stack([map_coordinates(a[:,:,i].astype(float),[q[:,:,1],q[:,:,0]],order=1,mode='constant',cval=0) for i in range(4)],axis=-1)
 return np.uint8(np.clip(out,0,255))
