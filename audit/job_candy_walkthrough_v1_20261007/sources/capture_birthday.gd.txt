var run_label := ""

var main: ReefMain

func _initialize() -> void:
	_run.call_deferred()

func _wait(count: int) -> void:
	for _i: int in range(count):
		await process_frame
		if recording:
			await RenderingServer.frame_post_draw
			_capture_motion()

func _state() -> Dictionary:
	if not is_instance_valid(world) or not is_instance_valid(world.surface):
		return {"context":"outside_candy_world","game":main.game if is_instance_valid(main) else "", "cake_mask":main.chapter2_cake_piece_mask if is_instance_valid(main) else 0}
	var s := world.surface as OperaGestureSurface
	var current: Dictionary = world.phases[world.phase_index] as Dictionary if world.phase_index >= 0 and world.phase_index < world.phases.size() else {}
	return {"phase":world.phase_index,"phase_name":current.get("name",""),
		"milestone":current.get("milestone",""),"phase_goal":current.get("goal",0),"mode":s.mode,"context":s.visual_context,
		"task_open":world.task_open,"phase_progress":world.phase_progress,"advance_pending":world.phase_advance_pending,
		"widget_fill":s.widget_fill,"crank_rotation":s.crank_rotation,"pour_tilt":s.pour_tilt,
		"pour_level":s.pour_level,"pour_reserve":s.pour_reserve,"held":s.held,"pointer":s.pointer_pos,
		"candy_sorted":s.candy_sorted,"candy_position":s.candy_position,"candy_type":s.candy_type,
		"target_placed":s.target_placed.duplicate(),"completion_accepted":s.completion_accepted,
		"draw_route":s.last_contextual_draw_route,"save_fixture_path":save_fixture_path,"pilot":false,"actor_visible":world.player_actor.visible,
		"actor_position":world.player_actor.position,"actor_size":world.player_actor.size,"actor_texture":world.player_actor.texture.resource_path,"actor_atlas":(world.player_actor.texture as AtlasTexture).atlas.resource_path if world.player_actor.texture is AtlasTexture else "","actor_atlas_region":(world.player_actor.texture as AtlasTexture).region if world.player_actor.texture is AtlasTexture else Rect2(),
		"surface_script":s.get_script().resource_path,"surface_position":s.position,"surface_size":s.size,"surface_modulate":s.modulate,
		"cake_mask":main.chapter2_cake_piece_mask,"persistent_cake_visible":is_instance_valid(world.chapter2_cake_scene) and world.chapter2_cake_scene.visible,"persistent_tray_visible":is_instance_valid(world.chapter2_cake_scene) and world.chapter2_cake_scene.candied_tray_sprite.visible,"panel_visible":world.action_panel.visible,"cake_rect":world.chapter2_cake_scene.get_global_rect() if is_instance_valid(world.chapter2_cake_scene) else Rect2(),"party_mask":main.chapter2_party_piece_mask,
		"visible_story_action":world.action_panel.get_meta("chapter2_candy_visible_action",""),
		"visible_story_berry_count":world.action_panel.get_meta("chapter2_candy_visible_berry_count",0)}

func _checkpoint() -> void:
	checkpoint_index += 1
	# Alternating flushed traces preserve the previous completed slot on ENOSPC.
	var path := OUT + "TRACE_%d.json" % (checkpoint_index % 2)
	var file := FileAccess.open(path, FileAccess.WRITE)
	assert(file != null, "Capture checkpoint open failed")
	file.store_string(JSON.stringify({"status":"PARTIAL_CHECKPOINT_REVIEW_PENDING","sequence":checkpoint_index,"time_msec":Time.get_ticks_msec(),"state":_state(),"motion_frames":motion_frames,"views":views,"events":events,"input_events":input_events},"\t"))
	file.flush()
	assert(file.get_error() == OK, "Capture checkpoint write failed")
	file.close()

func _event(value: Dictionary) -> void:
	events.append(value)
	_checkpoint()

func _input_event(value: Dictionary) -> void:
	input_events.append(value)
	_checkpoint()

func _capture_motion() -> void:
	var index := motion_frames.size()
	assert(index < 400, "Declared native-frame storage cap reached")
	var path := OUT+"native_frames/%s_%d_%04d.png" % [lane,width_now,index]
	var im := root.get_texture().get_image()
	assert(im.get_size() == Vector2i(width_now,720))
	assert(im.save_png(path)==OK)
	motion_frames.append({"path":path.trim_prefix("res://"),"sha256":FileAccess.get_sha256(path),"index":index,
		"width":width_now,"lane":lane,"time_msec":Time.get_ticks_msec(),"state":_state(),"direct_review":false,"scores":{},"owner_acceptance":null})
	if index % 12 == 0:
		_checkpoint()

func _view(tag: String) -> void:
	await RenderingServer.frame_post_draw
	var path := OUT+"native_views/%s_%d_%02d_%s.png" % [lane,width_now,views.size(),tag]
	var im := root.get_texture().get_image()
	assert(im.get_size() == Vector2i(width_now,720))
	assert(im.save_png(path)==OK)
	views.append({"path":path.trim_prefix("res://"),"sha256":FileAccess.get_sha256(path),"tag":tag,"width":width_now,
		"lane":lane,"state":_state() if is_instance_valid(world) and world.surface!=null else {},"direct_review":false,"scores":{}})
	_checkpoint()

func _touch(at: Vector2, pressed: bool) -> void:
	var e := InputEventScreenTouch.new()
	e.index=0
	e.position=at
	e.pressed=pressed
	Input.parse_input_event(e)
	_input_event({"kind":"down" if pressed else "up","at":at,"time_msec":Time.get_ticks_msec()})
	await _wait(2)

func _surface_touch(at: Vector2, pressed: bool) -> void:
	await _touch(world.surface.get_global_transform_with_canvas()*at,pressed)

func _drag(at: Vector2) -> void:
	var e := InputEventScreenDrag.new()
	e.index=0
	e.position=world.surface.get_global_transform_with_canvas()*at
	Input.parse_input_event(e)
	_input_event({"kind":"drag","at":e.position,"time_msec":Time.get_ticks_msec()})
	await _wait(2)

func _tap(c: Control) -> void:
	var at := c.get_global_transform_with_canvas()*(c.size*0.5)
	await _touch(at,true)
	await _touch(at,false)

func _open() -> void:
	for _i: int in range(400):
		var h := world._active_hotspot()
		if h!=null and h.visible:
			break
		await _wait(1)
	var h := world._active_hotspot()
	assert(h!=null and h.visible and not world.task_open)
	await _view("phase%d_invitation" % world.phase_index)
	await _tap(h.touch_button)
	for _i: int in range(400):
		if world.task_open:
			break
		await _wait(1)
	assert(world.task_open)
	await _wait(8)
	await _view("phase%d_task_open" % world.phase_index)

func _work() -> void:
	var phase := world.phase_index
	var s := world.surface as OperaGestureSurface
	var mode := s.mode
	var initial := _state()
	match mode:
		"bop":
			for _i: int in range(160):
				if world.phase_advance_pending or world.phase_index!=phase:
					break
				var target: Control = null
				for imp: Dictionary in world.combat_imps:
					if not bool(imp.get("popped",false)) and is_instance_valid(imp.get("node")):
						target=imp["node"] as Control
						break
				if target!=null:
					await _tap(target)
				else:
					await _wait(2)
		"pourt":
			await _surface_touch(s._pour_pitcher_hit_rect().get_center(),true)
			for _i: int in range(500):
				if world.phase_advance_pending or world.phase_index!=phase:
					break
				await _wait(1)
			await _surface_touch(s.pointer_pos,false)
		"choice":
			for _i: int in range(40):
				if world.phase_advance_pending or world.phase_index!=phase:
					break
				var at := Vector2((float(s.target_choice)+0.5)*s.size.x/float(s.choice_count),s.size.y*0.5)
				await _surface_touch(at,true)
				await _surface_touch(at,false)
		"circle":
			var centre := s._circle_pivot()
			var radius: float = 40.0 if story else minf(s.size.x,s.size.y)*0.35
			var before_passive := world.phase_progress
			await _wait(12)
			assert(is_equal_approx(world.phase_progress, before_passive))
			for _j: int in range(2):
				await _surface_touch(centre + Vector2(radius,0), true)
				await _wait(4)
				assert(is_equal_approx(world.phase_progress, before_passive))
				await _surface_touch(centre + Vector2(radius,0), false)
				await _wait(4)
				assert(is_equal_approx(world.phase_progress, before_passive))
			assert(not s.held)
			_event({"event":"PASSIVE_STATIONARY_REPEAT_AND_RELEASE_NO_PROGRESS", "progress":before_passive, "mode":s.mode})
			await _surface_touch(centre+Vector2(radius,0),true)
			for step: int in range(400):
				if world.phase_advance_pending or world.phase_index!=phase:
					break
				await _drag(centre+Vector2.from_angle(float(step+1)*TAU/32.0)*radius)
				if step == 16:
					var before_release := world.phase_progress
					await _surface_touch(s.pointer_pos, false)
					await _wait(12)
					assert(not s.held and is_equal_approx(world.phase_progress, before_release))
					_event({"event":"MID_ACTION_RELEASE_FREEZES_PROGRESS", "progress":before_release})
					await _surface_touch(centre + Vector2.from_angle(float(step+1)*TAU/32.0)*radius,true)
				if step==16 or step==32:
					await _view("phase%d_circle_step%d" % [phase,step])
			await _surface_touch(s.pointer_pos,false)
		"candy_sort":
			var no_input_progress := world.phase_progress
			await _wait(12)
			assert(is_equal_approx(world.phase_progress,no_input_progress))
			for repeated: int in range(2):
				await _surface_touch(s.candy_position,true)
				await _surface_touch(s.candy_position,false)
				assert(is_equal_approx(world.phase_progress,no_input_progress))
			var wrong_start := s.candy_position
			var wrong_bin := s._candy_bin_rect((s.candy_type+1)%3).get_center()
			await _surface_touch(wrong_start,true)
			for step: int in range(1,11):
				await _drag(wrong_start.lerp(wrong_bin,float(step)/10.0))
			await _surface_touch(wrong_bin,false)
			assert(is_equal_approx(world.phase_progress,no_input_progress) and s.candy_sorted == 0)
			_event({"event":"SORT_PASSIVE_REPEATED_STATIONARY_AND_WRONG_BIN_NO_PROGRESS","progress":no_input_progress,"count":s.candy_sorted,"qualification":"Production finger0 input at actual surface anchors; visible-vs-hit geometry acceptance remains separate."})
			for piece: int in range(16):
				if world.phase_advance_pending or world.phase_index!=phase:
					break
				var before_drop := s.candy_sorted
				var at := s.candy_position
				var to := s._candy_bin_rect(s.candy_type).get_center()
				await _surface_touch(at,true)
				for step: int in range(1,11):
					await _drag(at.lerp(to,float(step)/10.0))
				await _surface_touch(to,false)
				await _wait(6)
				assert(s.candy_sorted-before_drop in [0,1])
				_event({"event":"SORT_DISTINCT_DROP_ATTEMPT","attempt":piece,"before_count":before_drop,"after_count":s.candy_sorted,"progress":world.phase_progress,"accepted":s.candy_sorted>before_drop})
		"tap":
			for _i: int in range(16):
				if world.phase_advance_pending or world.phase_index!=phase:
					break
				var target := s._target_next_unplaced()
				var at := s._target_anchor_point(target) if target>=0 else s.size*0.5
				await _surface_touch(at,true)
				await _surface_touch(at,false)
				await _wait(4)
		_:
			push_error("Unimplemented current mode "+mode)
			quit(3)
			return
	if not (world.phase_advance_pending or world.phase_index!=phase):
		var failure_file := FileAccess.open(OUT+"FAILED_LIVE_STATE.json",FileAccess.WRITE)
		failure_file.store_string(JSON.stringify({"state":_state(),"events":events,"input_events":input_events,"window_size":root.size,"final_transform":root.get_final_transform()},"\t"))
		failure_file.close()
	assert(world.phase_advance_pending or world.phase_index!=phase)
	await _view("phase%d_earned_completion" % phase)
	_event({"event":"intentional_phase_completed","phase":phase,"mode":mode,"initial":initial,"after":_state()})
