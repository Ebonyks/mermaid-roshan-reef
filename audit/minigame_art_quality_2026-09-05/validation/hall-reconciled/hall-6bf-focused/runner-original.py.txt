from pathlib import Path
import hashlib, json, os, re, subprocess, sys, time
R = Path(__file__).resolve().parents[2]
O = R / 'tmp/hall-6bf-focused'
G = 'C:/Users/Peter/AppData/Local/Programs/MermaidReefTools/Godot/4.7.2/godot.exe'
PROBES = ['probe_opera_2d', 'probe_opera_diegetic_paths', 'probe_stage_navigation_2d', 'probe_chapter2_lawn', 'probe_opera_performance']

def main():
    O.mkdir(exist_ok=False)
    sources = {}
    for p in sorted((R/'scripts').rglob('*.gd')):
        sources[p.relative_to(R).as_posix()] = hashlib.sha256(p.read_bytes().decode('utf-8').replace('\r\n','\n').encode()).hexdigest()
    (O/'source-freeze.json').write_text(json.dumps(sources,indent=2),encoding='utf-8')
    results=[]
    for name in PROBES:
        profile=O/(name+'-profile'); profile.mkdir()
        env=os.environ.copy()
        for key,leaf in [('APPDATA','appdata'),('LOCALAPPDATA','localappdata'),('XDG_DATA_HOME','data'),('XDG_CONFIG_HOME','config')]:
            p=profile/leaf;p.mkdir();env[key]=str(p)
        command=[G,'--headless','--path',str(R),'-s','scripts/'+name+'.gd']
        start=time.monotonic()
        with (O/(name+'.log')).open('wb') as f:
            run=subprocess.run(command,cwd=R,env=env,stdout=f,stderr=subprocess.STDOUT)
        raw=(O/(name+'.log')).read_bytes()
        errors=re.findall(rb'(?im)^(?:SCRIPT ERROR|ERROR:|.*\|FAIL|.*\bFAIL\b)',raw)
        results.append({'probe':name,'exit':run.returncode,'hard_error_lines':[x.decode('utf-8','replace') for x in errors],
            'seconds':round(time.monotonic()-start,3),'log_sha256':hashlib.sha256(raw).hexdigest(),'command':command})
        (O/'results.json').write_text(json.dumps(results,indent=2),encoding='utf-8')
        print(name,'EXIT',run.returncode,'ERRORS',len(errors),flush=True)
        if run.returncode or errors:return 1
    unchanged=all(hashlib.sha256((R/p).read_bytes().decode('utf-8').replace('\r\n','\n').encode()).hexdigest()==h for p,h in sources.items())
    (O/'source-freeze-check.json').write_text(json.dumps({'unchanged':unchanged,'sources':len(sources)}),encoding='utf-8')
    return 0 if unchanged else 2
if __name__=='__main__':sys.exit(main())
