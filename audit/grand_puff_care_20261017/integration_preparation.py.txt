from pathlib import Path
import subprocess, json, sys
root = Path(__file__).resolve().parents[2]
out = root / 'audit/grand_puff_care_20261017'
assert not subprocess.check_output(['git', 'diff', '--name-only', '--diff-filter=U'], cwd=root).strip()
care_files = ['scripts/audio_director.gd', 'scripts/dust_boss_lesson_2d.gd', 'scripts/dust_bunny_boss_sprite.gd', 'scripts/games/dust_boss.gd', 'scripts/grand_puff_care_2d.gd', 'scripts/main.gd', 'scripts/probe_audio.gd', 'scripts/probe_day_one_story_clips.gd', 'scripts/probe_dust_boss.gd']

pin_path = out / 'source_pin_integration.json'
pin = json.loads(pin_path.read_text(encoding='utf-8'))

source = (root / 'tmp/puff-crash/run_care_v2_ci.py').read_text(encoding='utf-8')

    ('source_pin_final_v2.json', 'source_pin_integration.json'),
    ('care-full-final-v2-ci.log', 'care-integration-ci.log'),
    ('full_final_v2_ci_result.json', 'integration_ci_result.json'),
    ('full_final_v2_ci.log', 'integration_ci.log'),
    ('FULL_FINAL_V2_CI', 'INTEGRATION_CI'),

    source = source.replace(old, new)
(root / 'tmp/puff-crash/run_care_integration_ci.py').write_text(source, encoding='utf-8')
(out / 'integration_ci_runner.py.txt').write_text(source, encoding='utf-8')
(out / 'integration_resolution.py.txt').write_bytes((root / 'tmp/puff-crash/resolve_care_dev_appends.py').read_bytes())

Original V2 full pass is retained separately. All decision content preserved; published incoming register sequence remains the append-only prefix.'})
impact_path.write_text(json.dumps(impact, indent=2, ensure_ascii=False) + '\n', encoding='utf-8')
