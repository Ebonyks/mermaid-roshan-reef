from pathlib import Path
import subprocess, json, hashlib

root = Path(__file__).resolve().parents[2]
impact = json.loads((root / 'design/audit_impacts/grand-puff-care-20261017.json').read_text(encoding='utf-8'))
paths = impact['files'] + ['design/audit_impacts/grand-puff-care-20261017.json']
subprocess.run(['git', 'add', '-f', '--', *paths], cwd=root, check=True)
index = {}
for line in subprocess.check_output(['git', 'ls-files', '--stage'], cwd=root, text=True).splitlines():
    fields, path = line.split('\t', 1)
    mode, oid, stage = fields.split()
    assert stage == '0'
    index[path] = oid
pin = json.loads((root / 'audit/grand_puff_care_20261017/source_pin_final_v2.json').read_text(encoding='utf-8'))
rows = []
for item in pin['source_files']:
    path = item['path']
    assert hashlib.sha256((root / path).read_bytes()).hexdigest() == item['sha256'], path
    oid = subprocess.check_output(['git', 'hash-object', '--path=' + path, path], cwd=root, text=True).strip()
    assert index[path] == oid, path
    rows.append({'path': path, 'local_raw_sha256': item['sha256'], 'git_blob_oid': oid})
for folder in ['assets_src/animation/grand_puff_care_20261017', 'assets_src/audio/grand_puff_care_20261017']:
    for path in sorted((root / folder).rglob('*')):
        if not path.is_file() or path.relative_to(root).as_posix() not in index:
            continue
        rel = path.relative_to(root).as_posix()
        blob = subprocess.check_output(['git', 'cat-file', 'blob', index[rel]], cwd=root)
        assert blob == path.read_bytes(), rel
receipt = {'status': 'PASS', 'source_pin': 'audit/grand_puff_care_20261017/source_pin_final_v2.json',
    'scope': 'Every frozen local source maps through its existing Git path filters to the staged Git blob. Text checkout line endings may differ; native animation/audio source blobs are separately byte-identical to local files under scoped -text attributes. No runtime change during full suite.',
    'source_files': rows, 'source_count': len(rows)}
(root / 'audit/grand_puff_care_20261017/staged_source_binding.json').write_text(json.dumps(receipt, indent=2) + '\n', encoding='utf-8')
print('CARE_INDEX_BINDING|PASS|' + str(len(rows)))
