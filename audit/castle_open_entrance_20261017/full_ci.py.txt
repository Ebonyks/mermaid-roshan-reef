from pathlib import Path
import subprocess,hashlib,json,re,time
from datetime import datetime,timezone
root=Path(__file__).resolve().parents[1]
out=root/'audit/castle_open_entrance_20261017'
source_paths=sorted(p for folder in ['scripts','scenes','tools','shaders'] for p in (root/folder).rglob('*') if p.is_file() and '__pycache__' not in p.parts)
source_paths += [root/'scripts/arena/sky_lagoon_layout.json',root/'assets/sprites/sky_lagoon/sky_lagoon_castle_open_v1.png']
source={p.relative_to(root).as_posix():hashlib.sha256(p.read_bytes()).hexdigest() for p in source_paths}
started=datetime.now(timezone.utc).isoformat();t=time.monotonic()
cmd=['C:/Program Files/Git/bin/bash.exe','--noprofile','--norc',str(root/'tmp/castle_open_full_ci.sh')]
with (out/'full_ci.log').open('w',encoding='utf8') as log:
 result=subprocess.run(cmd,cwd=root,stdout=log,stderr=subprocess.STDOUT)
text=(out/'full_ci.log').read_text(encoding='utf8',errors='replace')
probes=re.findall(r'^PROBE (\S+) process exit: (\d+)$',text,re.M)
expected=re.search(r'for p in (.+?); do',(root/'scripts/ci.sh').read_text()).group(1).split()
drift=[name for name,value in source.items() if not (root/name).is_file() or hashlib.sha256((root/name).read_bytes()).hexdigest()!=value]
receipt={'command':cmd,'started_utc':started,'finished_utc':datetime.now(timezone.utc).isoformat(),'elapsed_seconds':time.monotonic()-t,'process_exit':result.returncode,'expected_probes':len(expected),'completed_probes':probes,'source_drift':drift,'source_sha256':source,'log_sha256':hashlib.sha256((out/'full_ci.log').read_bytes()).hexdigest(),'status':'PASS' if result.returncode==0 and len(probes)==len(expected) and all(code=='0' for _,code in probes) and not drift else 'FAIL'}
(out/'full_ci.json').write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf8')
print(json.dumps({k:v for k,v in receipt.items() if k not in ['source_sha256','completed_probes']}))
raise SystemExit(0 if receipt['status']=='PASS' else 1)
