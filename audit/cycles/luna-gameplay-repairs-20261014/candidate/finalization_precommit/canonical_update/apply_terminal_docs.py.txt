def require(condition: bool, message: str) -> None:
    if not condition:
        raise RuntimeError(message)


def digest(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def read_json(path: Path):
    return json.loads(path.read_text(encoding="utf-8-sig"))


def confined(path: Path, base: Path) -> Path:
    path = path.resolve()
    require(path.is_relative_to(base.resolve()), "Path outside authorized base: " + str(path))
    return path


def git(*args: str) -> bytes:
    result = subprocess.run(["git", *args], cwd=WORKTREE, capture_output=True, check=False)
    require(result.returncode == 0, "Read-only git command failed: " + repr(args))
    return result.stdout


def changed_paths() -> set[str]:
    changed = git("diff", "--name-only", "-z", "HEAD", "--")
    untracked = git("ls-files", "--others", "--exclude-standard", "-z")
    names = {s.decode("utf-8").replace("\\", "/") for s in (changed + untracked).split(b"\0") if s}
    require(all(not Path(s).is_absolute() and ".." not in Path(s).parts for s in names),
            "Unexpected Git path")
    return names - {IMPACT}


def verify_archive(folder: Path, expected_sha: str) -> dict:
    manifest_path = folder / "MANIFEST.json"
    require(HEX.fullmatch(expected_sha) is not None, "Expected manifest SHA must be explicit")
    require(digest(manifest_path.read_bytes()) == expected_sha, "Archive manifest hash differs from reviewed hash")
    manifest = read_json(manifest_path)
    rows = manifest.get("files")
    require(isinstance(rows, list) and rows, "Archive payload list absent")
    names = set()
    for row in rows:
        name = row.get("path")
        require(isinstance(name, str) and name not in names, "Duplicate/invalid archive path")
        names.add(name)
        path = confined(folder / name, folder)
        require(path.is_file() and HEX.fullmatch(row.get("sha256", "")) is not None, "Invalid archive payload")
        data = path.read_bytes()
        require(digest(data) == row["sha256"] and len(data) == row["bytes"], "Archive payload bytes differ: " + name)
    actual = {p.relative_to(folder).as_posix() for p in folder.rglob("*") if p.is_file()}
    require(actual == names | {"MANIFEST.json"}, "Archive has missing or unlisted payloads")
    require(manifest.get("payload_count", manifest.get("file_count")) == len(rows), "Archive count differs")
    return manifest


def normalized(raw: bytes) -> tuple[str, str]:
    text = raw.decode("utf-8")
    if "\r\n" in text:
        require(text.count("\r\n") == text.count("\n"), "Mixed newline conventions in target")
        return text.replace("\r\n", "\n"), "\r\n"
    return text, "\n"


def once(text: str, old: str, new: str) -> str:
    require(text.count(old) == 1, "Expected exactly one scoped replacement: " + old)
    return text.replace(old, new, 1)


def paragraph(text: str, starts: str, old: str, new: str) -> str:
    require(text.count(starts) == 1, "Paragraph anchor is not unique")
    begin = text.index(starts)
    end = text.find("\n\n", begin)
    if end < 0:
        end = len(text)
    return text[:begin] + once(text[begin:end], old, new) + text[end:]


def link_for(document: str, target: Path) -> str:
    return os.path.relpath(target, (WORKTREE / document).parent).replace("\\", "/")
