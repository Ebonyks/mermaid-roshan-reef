def sources():
    # Include committed, modified and unignored new files in all runtime areas.
    names = subprocess.check_output(['git', 'ls-files', '-z', '--cached', '--others',
        '--exclude-standard', '--', 'scripts', 'scenes', 'shaders', 'resources',
        'tools', 'project.godot', 'default_bus_layout.tres', 'assets',
        'assets_src/audio/objective_guides_20261014'], cwd=root).decode('utf-8').split('\0')
    return {name: hashlib.sha256((root / name).read_bytes()).hexdigest()
        for name in sorted(set(names)) if name and (root / name).is_file()}


def process_snapshot():
    # Process IDs, parent IDs and creation times only; never command lines.
    command = [
        'powershell.exe', '-NoProfile', '-NonInteractive', '-Command',
        "$ErrorActionPreference = 'Stop'; "
        "@(Get-CimInstance Win32_Process | Where-Object { $null -ne $_.CreationDate } | ForEach-Object { "
        "[pscustomobject]@{pid=[int]$_.ProcessId; parent_pid=[int]$_.ParentProcessId; "
        "created_utc=$_.CreationDate.ToUniversalTime().ToString('o')} "
        "}) | ConvertTo-Json -Compress",
    ]
    result = subprocess.run(command, capture_output=True, text=True,
        encoding='utf-8', errors='replace', timeout=15,
        creationflags=subprocess.CREATE_NO_WINDOW, check=True)
    values = json.loads(result.stdout)
    if isinstance(values, dict):
        values = [values]
    return {value['pid']: value for value in values}


def owned_tree(snapshot, child_pid):
    owned = {child_pid}
    while True:
        descendants = {pid for pid, value in snapshot.items()
            if value['parent_pid'] in owned}
        expanded = owned | descendants
        if expanded == owned:
            return {pid: snapshot[pid] for pid in sorted(owned) if pid in snapshot}
        owned = expanded


def stop_owned_tree(process):
    result = {'method': 'taskkill /PID <owned-child> /T /F',
        'owned_child_pid': process.pid, 'stopped_verified': False, 'errors': []}
    tree = {}
    try:
        tree = owned_tree(process_snapshot(), process.pid)
        result['observed_owned_processes'] = list(tree.values())
        if process.pid not in tree:
            result['errors'].append('Owned child absent from pre-kill process snapshot')
    except Exception as error:
        result['errors'].append('Pre-kill process snapshot: ' + str(error))
    try:
        if process.poll() is not None:
            # Do not target a recycled PID after the owned Popen has exited.
            result['errors'].append('Owned child exited before taskkill; tree cleanup unconfirmed')
        else:
            killed = subprocess.run(['taskkill.exe', '/PID', str(process.pid), '/T', '/F'],
                capture_output=True, text=True, encoding='utf-8', errors='replace',
                timeout=30, creationflags=subprocess.CREATE_NO_WINDOW)
            result.update(taskkill_exit_code=killed.returncode,
                taskkill_stdout=killed.stdout, taskkill_stderr=killed.stderr)
            if killed.returncode != 0:
                result['errors'].append('taskkill did not report success')
    except Exception as error:
        result['errors'].append('taskkill: ' + str(error))
    try:
        result['owned_child_exit_code'] = process.wait(timeout=15)
    except subprocess.TimeoutExpired:
        result['errors'].append('Owned child still running after tree kill deadline')
    except Exception as error:
        result['errors'].append('Owned child stopped verification: ' + str(error))
    try:
        remaining = process_snapshot()
        survivors = [pid for pid, value in tree.items()
            if remaining.get(pid, {}).get('created_utc') == value['created_utc']]
        result['surviving_owned_pids'] = survivors
        if survivors:
            result['errors'].append('Observed owned processes survived tree kill')
    except Exception as error:
        result['errors'].append('Post-kill process snapshot: ' + str(error))
    result['stopped_verified'] = bool(tree) and not result['errors']
    return result

def classify_failure_lines(output, kind, exit_code, attributed):
    raw = [line for line in output.splitlines() if re.search(
        r'(^|[| :])FAIL([| :]|$)|SCRIPT ERROR|Parse Error|Compile Error', line)]
    known_count = raw.count(KNOWN_SELFTEST_FAILURE)
    allow = kind == 'ci' and exit_code == 0 and attributed and known_count == 1
    ignored = [KNOWN_SELFTEST_FAILURE] if allow else []
    blocking = [line for line in raw if not (allow and line == KNOWN_SELFTEST_FAILURE)]
    return {'raw': raw, 'ignored': ignored, 'blocking': blocking,
        'known_selftest_count': known_count, 'allowance_applied': allow}
