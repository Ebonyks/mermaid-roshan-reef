extends "res://scripts/probe_opera_art.gd"
## Diagnostic: real specialist/world signals, injected current story phase data.
func _run() -> void:
	if not await _set_aspect(Vector2i(1280, 720)) or not await _start_runtime():
		quit(1)
		return
	var room_id := CastleCareerRoutes.room_for_act(2)
	if not await _prepare_career_route(room_id, 2):
		quit(1)
		return
	routes._launch(room_id, 2)
	if not await _wait_career_ready(2, room_id):
		quit(1)
		return
	var opera := main.opera_game as OperaHouse
	var current := opera.act as OperaAct
	current.set_process(false)
	var world := current.career_world_2d
	world.set_process(false)
	world.phases = ChapterTwoCareerSceneAdapter.PHASE_SETS["ballerina"]["phases"].duplicate(true)
	for phase_id in [1, 2]:
		world.active = true
		world.phase_index = phase_id
		world._show_phase()
		world.phase_gap = 0.0
		var ballet := world.surface as OperaBalletSurface
		ballet.set_process(false)
		ballet._finish_demo()
		if phase_id == 1:
			ballet._press(ballet.ribbon_point(0.0))
			for step in range(1, 401):
				ballet._drag(ballet.ribbon_point(float(step) / 400.0))
		else:
			ballet._press(ballet.twirl_center() + Vector2.RIGHT * ballet.twirl_radius())
			for step in range(1, 401):
				ballet._drag(ballet.twirl_center() + Vector2.from_angle(float(step) / 200.0 * TAU) * ballet.twirl_radius())
		print("BALLET_DIAGNOSTIC|phase=", phase_id, "|goal=", world.phases[phase_id]["goal"], "|world_progress=", world.phase_progress, "|surface_complete=", ballet._mode_complete(), "|input_blocked=", ballet._input_blocked(), "|completion_hold=", world.phase_complete_t)
	await _teardown_runtime()
	quit(0)
